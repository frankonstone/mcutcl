cmake_minimum_required(VERSION 3.25)
project(mcutcl VERSION 0.2.0 DESCRIPTION "Project-owned Tcl 9 interpreter" LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(MCUTCL_BUILD_TESTS "Build interpreter tests" ON)
option(MCUTCL_BUILD_CLI "Build mcutcl CLI" ON)
option(MCUTCL_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
set(MCUTCL_MCUTEST_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../mcutest" CACHE PATH "mcutest source tree")
set(MCUTCL_MCUCLI_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../mcucli" CACHE PATH "mcucli source tree")
function(mcutcl_options target)
    target_compile_features(${target} PUBLIC cxx_std_23)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /Zc:preprocessor /EHs-c- /GR-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
        if(MCUTCL_ENABLE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
add_library(mcutcl_runtime STATIC src/interpreter.cpp src/list.cpp)
add_library(mcutcl::mcutcl ALIAS mcutcl_runtime)
set_target_properties(mcutcl_runtime PROPERTIES EXPORT_NAME mcutcl)
target_include_directories(mcutcl_runtime PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
mcutcl_options(mcutcl_runtime)
if(MCUTCL_BUILD_CLI)
    find_package(mcucli CONFIG QUIET)
    if(NOT TARGET mcucli::mcucli AND EXISTS "${MCUTCL_MCUCLI_SOURCE_DIR}/CMakeLists.txt")
        set(MCUCLI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        set(MCUCLI_INSTALL OFF CACHE BOOL "" FORCE)
        set(MCUCLI_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
        add_subdirectory("${MCUTCL_MCUCLI_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/_deps/mcucli" EXCLUDE_FROM_ALL)
    endif()
    if(NOT TARGET mcucli::mcucli)
        message(FATAL_ERROR "Provide mcucli or disable MCUTCL_BUILD_CLI; downloads are not automatic")
    endif()
    add_executable(mcutcl src/main.cpp)
    target_link_libraries(mcutcl PRIVATE mcutcl::mcutcl mcucli::mcucli)
    mcutcl_options(mcutcl)
    install(TARGETS mcutcl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS mcutcl_runtime EXPORT mcutcl-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT mcutcl-targets NAMESPACE mcutcl:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/mcutcl)
configure_package_config_file(cmake/mcutcl-config.cmake.in "${CMAKE_BINARY_DIR}/mcutcl-config.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/mcutcl)
write_basic_package_version_file("${CMAKE_BINARY_DIR}/mcutcl-config-version.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_BINARY_DIR}/mcutcl-config.cmake" "${CMAKE_BINARY_DIR}/mcutcl-config-version.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/mcutcl)
if(MCUTCL_BUILD_TESTS)
    enable_testing()
    find_package(mcutest CONFIG QUIET)
    if(NOT TARGET mcutest::mcutest AND EXISTS "${MCUTCL_MCUTEST_SOURCE_DIR}/CMakeLists.txt")
        set(MCUTEST_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        set(MCUTEST_INSTALL OFF CACHE BOOL "" FORCE)
        add_subdirectory("${MCUTCL_MCUTEST_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/_deps/mcutest" EXCLUDE_FROM_ALL)
    endif()
    if(NOT TARGET mcutest::mcutest)
        message(FATAL_ERROR "Provide mcutest or disable MCUTCL_BUILD_TESTS")
    endif()
    add_executable(mcutcl_tests tests/main.cpp tests/interpreter.cpp tests/lists.cpp)
    target_link_libraries(mcutcl_tests PRIVATE mcutcl::mcutcl mcutest::mcutest)
    mcutest_configure_target(mcutcl_tests MAX_TESTS 256 SOURCE_LOCATION ON)
    mcutcl_options(mcutcl_tests)
    add_test(NAME interpreter COMMAND mcutcl_tests)
    add_custom_target(evidence COMMAND ${CMAKE_COMMAND}
        "-DTEST_BINARY=$<TARGET_FILE:mcutcl_tests>" "-DOUTPUT_PATH=${CMAKE_BINARY_DIR}/artifacts/mcutest.json"
        -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/CollectTests.cmake DEPENDS mcutcl_tests VERBATIM)
    if(MCUTCL_BUILD_CLI)
        add_test(NAME cli_eval COMMAND mcutcl eval --script "set x {two words}; set y \"value=$x\"")
        set_tests_properties(cli_eval PROPERTIES PASS_REGULAR_EXPRESSION "value=two words")
        add_test(NAME cli_error COMMAND mcutcl eval --script "unimplemented")
        set_tests_properties(cli_error PROPERTIES WILL_FAIL TRUE)
    endif()
endif()
